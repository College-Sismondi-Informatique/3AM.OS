\section{Théorème de Bézout et PPMC}

\begin{theoreme}[\textit{Théorème de Bézout}]%\label{coefbezout}
Soient $a;b \in \Z$. Alors il existe des entiers $\alpha$ et $\beta$ tels que
$$\pgdc(a;b)=\alpha\fois a + \beta\fois b$$
\end{theoreme}


\begin{demonstration}
Soient $a, b \in \mathbb{N}^*$. Considérons l'ensemble $E = \{ am + bn > 0 \mid m, n \in \mathbb{Z} \} \subset \mathbb{N}^*$ .

On remarque que $a$ et $b$ appartiennent à $E$ car $a = a \cdot 1 + b \cdot 0 > 0$ et $b = a \cdot 0 + b \cdot 1 > 0$.

Donc $E$ est non vide et $E\subset \mathbb{N}^*$. D'après le principe du bon ordre (tout ensemble non vide d'entiers naturels possède un plus petit élément), $E$ admet un plus petit élément strictement positif.
Notons
De plus, $E$ est un ensemble non vide dont chaque élément est plus grand que $0$. Il contient donc un plus petit élément strictement positif (car plus grand que $0$) notons-le $d = au + bv$.

\begin{itemize}
	\item \textbf{Démontrons que $pgcd(a;b) \le d$ :}
	
	Par définition $pgcd(a;b)$ divise $a$ et $b \underset{\text{Prop. \ref{prop_div_comb_lin}}}{\Rightarrow} pgcd(a;b)$ divise $au + bv = d$
	
	On a donc $pgcd(a;b) \le d$.
	
	\item \textbf{Démontrons que $pgcd(a;b) \ge d$ :}
	
	On effectue la division euclidienne de $a$ par $d$ et on obtient un unique couple d'entiers $(q;r)$ tel que $a = dq + r$ avec $0 \le r < d$. 
	
	Mais on a $r = a - dq = a - (au + bv)q = a - auq - bvq = a \cdot (1 - uq) + b \cdot (-vq)$
	
	Par conséquent, la seule possibilité est d'avoir $r = 0$, car sinon on a $0 < r < d$ ce qui voudrait dire que $r \in E$, mais par construction on ne peut pas avoir $r < d$, car $d$ est le plus petit élément de $E$.
	
	Comme $r = 0$, on a que $d$ divise $a$. Similairement, on peut montrer de même que $d$ divise $b$.
	
	Comme $d$ divise $a$ et $d$ divise $b$ on a $d \le pgcd(a;b)$.
	
	\item \textbf{Pour finir}, la seule possibilité est $d = pgcd(a;b)$ et donc il existe bien deux entiers relatifs $u$ et $v$ tels que $a \cdot u + b \cdot v = pgcd(a;b)$.
	
\end{itemize}
\vspace{-3mm}\hfill $\blacksquare$
\end{demonstration}	


\medskip

\begin{remarques}
	\item Les entiers $\alpha$ et $\beta$ s'appellent les \textbf{coefficients de Bézout} et ils peuvent s’obtenir en \guillemets{remontant} l'algorithme d'Euclide.
	
	\item Il faut noter que les entiers $\alpha$ et $\beta$ ne sont pas uniques.
\end{remarques}


\bigskip

 Calculons les coefficients de Bézout pour $a=600$ et $b=124$.

\pagebreak
\begin{corolaire}\label{cor:diviseur_pgcd}
Si $d|a$ et $d|b$ alors $d | \pgdc(a,b)$.
\end{corolaire}

\smallskip
\begin{exemple} $4|16$ et $4|24$ donc $4$ divise $\pgdc(16,24)=8$.
\end{exemple}

\smallskip
\begin{demonstration}
Par le théorème de Bézout, il existe $\alpha,\,\beta\in\N$ tels que $\alpha\fois a + \beta\fois b=\pgdc(a,b)$.

\medskip
Comme $d|a$ et $d|b$ alors $d | (\alpha\fois a+\beta\fois b)=\pgdc(a,b)$ (combinaison linéaire).
\vspace{-3mm}\hfill $\blacksquare$
\end{demonstration}

\bigskip
\begin{corolaire}
	\label{cor-Bezout}
Soient $a, b$ deux entiers. $a$ et $b$ sont premiers entre eux
\textbf{si et seulement si} il existe $u,v \in \Z$ tels que
$$ua+vb=1$$
\end{corolaire}

\begin{demonstration}
Le sens direct $(\Rightarrow)$ est une conséquence du corolaire \ref*{cor:diviseur_pgcd}.

\medskip
Pour le sens indirect $(\Leftarrow)$, supposons qu'il existe $u,v\in\Z$ tels que $ua+vb=1$.

\smallskip
Comme $\pgdc(a,b)|a$ alors $\pgdc(a,b)|au$. De même $\pgdc(a,b)|bv$.

\smallskip
Donc $\pgdc(a,b)|(au+bv)=1$. Donc $\pgdc(a,b)=1$.

\vspace{-3mm}\hfill $\blacksquare$
\end{demonstration}

\medskip
\begin{remarque}
Si on trouve deux entiers $u',v'$ tels que $u'a+v'b=d$, cela n'implique
\textbf{pas} que $d=\pgdc(a,b)$. On sait seulement alors que $\pgdc(a,b)|d$.

\smallskip
Par exemple si $a=12$, $b=8$ alors $12 \cdot 1 + 8 \cdot 3 = 36$ et $\pgdc(a,b)=4$.
\end{remarque}

\bigskip
\begin{corolaire}[\normalsize Lemme de Gauss]
\index{lemme!de Gauss}
Soient $a,b,c \in \Z$.

$$\textrm{Si} \ \ a | bc \ \  \textrm{\bf et} \ \  \pgdc(a,b)=1 \ \  \textrm{alors} \ \  a|c$$
\end{corolaire}

\medskip
\begin{exemple} si $4 | (7c)$, comme $4$ et $7$ sont premiers entre eux, alors $4|c$.
\end{exemple}

\begin{demonstration}
Comme $\pgdc(a,b)=1$, alors il existe $u,v\in \Z$ tels que
$ua+vb=1$.

\smallskip
On multiplie cette égalité par $c$ pour obtenir
$acu + bcv = c$.

\smallskip
Mais $a|acu$ et par hypothèse $a|bcv$ donc
$a$ divise $acu+bcv=c$.

\vspace{-3mm}\hfill $\blacksquare$
\end{demonstration}

\subsection*{Le plus petit multiple commun}

\begin{definition}
Le \textbf{plus petit multiple commun} de deux nombres $a$ et $b$, noté $\ppmc(a,b)$,
est le plus petit entier positif divisible à la fois par $a$ et par $b$.
\end{definition}

\begin{exemple}$\ppmc(12,9)=36$.
\end{exemple}

Le $\pgdc$ et le ppmc sont liés par la formule suivante :

\medskip
\begin{proposition}
Si $a,b$ sont des entiers (non tous les deux nuls) alors,

\begin{center}
\fbox{$\pgdc(a,b) \cdot  \ppmc(a,b) = |ab|$}
\end{center}
\end{proposition}

\begin{demonstration}
Sans perte de généralité, on peut supposer $a>0$ et $b>0$.

\smallskip
Posons $d= \pgdc(a,b)$ et définissons $m= \frac{|ab|}{\pgdc(a,b)}$.

\bigskip
On écrit $a=da'$ et $b=db'$ avec $\pgdc(a',b')=1$. Donc 
$$ab=d^2 a'b'\eh{4}\textrm{et par suite}\eh{4}m=da'b'$$
Ainsi $m=ab'=a'b$ est un multiple de $a$ et de $b$.

\medskip
Il reste à montrer que c'est le plus petit multiple. Si $n$ est un autre multiple
de $a$ et de $b$ alors 

$$n= ka= \ell b$$

donc $$kda'=\ell d b' \eh{4}\textrm{et par conséquent}\eh{4} ka'=\ell b'$$

La dernière égalité implique que $a' | \ell b'$. Mais comme $\pgdc(a',b')=1$ , le lemmes de Gauss assure que $a'|\ell$ et donc $a'b | \ell b$.

\medskip
Comme $m=a'b$ et $\ell b = n$, nous avons que $m|n$.

\medskip
Nous avons montré que tout multiple commun de $a$ et $b$ est divisible par $m$, donc $m=\ppmc(a,b)$ et finalement 
$$\pgdc(a,b) \cdot  \ppmc(a,b) = |ab|$$

\vspace{-3mm}\hfill $\blacksquare$
 
\end{demonstration}

Voici un autre résultat concernant le ppmc qui se démontre en utilisant la décomposition en facteurs premiers (que nous étudierons à la section suivante):
\begin{proposition}
Si $a|c$ et $b|c$ alors $\ppmc(a,b)|c$.
\end{proposition}

\begin{remarque}
Il serait faux de penser que $ab|c$.

\smallskip
Par exemple $6|36$, $9|36$ mais $6\cdot 9$ ne divise pas $36$.
Par contre $\ppmc(6,9)=18$ divise bien $36$.
\end{remarque}


\subsection*{L'inverse dans $\mathbb{Z}_n$}
\textbf{Question} : Comment déterminer si un élément $\cl{a} \in  \mathbb{Z}_n$ admet un inverse ? \\

On cherche un élément $\cl{b}$ tel que $$\cl{a}\cdot \cl{b} = \cl{a\cdot b} = \cl{1}$$

Ce qui est équivalent à trouver un $b \in \mathbb{Z}$ tel que 
$$a \cdot b \equiv 1 \pmod n \ssi a \cdot b = q\cdot n + 1 \text{ avec } q\in \mathbb{Z} $$
La dernière égalité peut se réécrire $ a \cdot b + (-q)\cdot n + 1 $, ce qui ressemble fortement au corollaire \ref{cor-Bezout} du théorème de Bézout.

\begin{proposition}
  Soit $\cl{a} \in \mathbb{Z}_n$ où $a$ est un représentant principal ( $0 \leq a<n$ ).
L'élément $\bar{a}$ possède un inverse si et seulement si $a$ et $n$ sont premiers entre eux (c'est-à-dire si $pgcd(a ; n)=1$ ).
\end{proposition}

\medskip

\begin{demonstration}
$$
\begin{aligned}
	\operatorname{pgcd}(a ; n)=  1 &\stackrel{\text {Cor. \ref{cor-Bezout} }}{\Leftrightarrow} \exists(u ; v) \in \mathbb{Z}^2 \text { tel que au }+n v=1 \\
	& \Leftrightarrow a \cdot u=n \cdot(-v)+1 \text { avec }-v \in \mathbb{Z} \\
	& \Leftrightarrow a \cdot u \equiv 1 \bmod n \\
	& \Leftrightarrow \cl{a} ^{-1} = \cl{u}
\end{aligned}
$$
\vspace{-3mm}\hfill $\blacksquare$
\end{demonstration}	

\medskip

\begin{remarque}
	Pour calculer l'inverse d'un élément $\cl{a} \in \mathbb{Z}_n$, il suffit de calculer les coefficients de Bézout.
\end{remarque}


\medskip

\begin{exemple}
	On veut calculer $\cl[26]{5} ^{-1}$ dans $\mathbb{Z}_{26}$.\\
	

		On calcule le $pgcd(5;26)$ à l'aide de \textit{l'algorithme d'Euclide} afin de vérifier si $\cl[26]{5}$ admet un inverse. Si le $pgcd(5;26) = 1$ alors on calcule \textit{les coefficients de Bézout} à l'aide de \textit{l'algorithme d'Euclide (étendu)}.
		
		\begin{itemize}[label=$\bullet$]
			\item $26 = 5 \cdot 5 + \fbox{1} \quad \Rightarrow \quad \fbox{1} = 5 \cdot \underbrace{(-5)}_{=u} + 26 \cdot \underbrace{(1)}_{=v}$
			\item $5 = 1 \cdot 5 + 0$
		\end{itemize}
		
		Comme $pgcd(5;26) = 1$ on a que $\cl[26]{5}$ possède un inverse.
		
		On obtient alors $5 \cdot (-5) = 26 \cdot (-1) + 1 \quad 0 \le 1 < 26 \quad \Leftrightarrow 5 \cdot (-5) \equiv 1 \bmod 26$
		
		Donc on a que $\cl[26]{5} ^{-1}= \cl[26]{-5}$.
		
		Si on souhaite exprimer l’inverse avec un représentant principal, on effectue une \textit{division Euclidienne} de $-5$ par $26$ :
		\[
		-5 = 26 \cdot (-1) + 21 \quad 0 \le 21 < 26 \quad \text{donc } 21 \equiv -5 \bmod 26
		\]
		
		\textbf{Conclusion} : $\cl[26]{5} ^{-1}= \cl[26]{21}$

\end{exemple}
