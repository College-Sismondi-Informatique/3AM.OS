\section{Congruences}




\begin{definition}
Soit $n \ge 2$ un entier. On dit que $a$ est congru à $b$ modulo $n$,
si $n$ divise $b-a$. Dans ce cas on note :
$$a \equiv b \pmod n.$$

En d'autres termes : \eh{1}$a \equiv b \pmod n \quad \ssi \quad \exists k \in \Z \eh{2}\textrm{tel que}\eh{2} a = b +kn$.
\end{definition}


Remarquons que $n$ divise $a$ si et seulement si $a\equiv 0 \pmod n$.

\vspace{3mm}

\begin{exemples}{1mm}
	\item $59 \equiv 17 \pmod 7 \quad \text{car} \quad 59 = 7 \cdot 8 + \boxed{3} \quad \text{et} \quad 17 = 7 \cdot 2 + \boxed{3}$
	
	$59 \text{ est congru à } 17 \text{ modulo } 7 \text{ car ils ont le même reste.}$
	
	\item $3 \equiv -1 \pmod 2 \quad \text{car} \quad 3 = 2 \cdot 1 + \boxed{1} \quad \text{et} \quad -1 = 2 \cdot (-1) + \boxed{1}$
	
	\item$-14 \equiv -2 \pmod 4 \quad \text{car} \quad -14 = 4 \cdot (-4) + \boxed{2} \quad \text{et} \quad -2 = 4 \cdot (-1) + \boxed{2}$
	
	\item $-14 \not\equiv -2 \pmod 5 \quad \text{car} \quad -14 = 5 \cdot (-3) + \boxed{1} \quad \text{et} \quad -2 = 5 \cdot (-1) + \boxed{3}$
	
	$-14 \text{ n'est pas congru à } -2 \text{ modulo } 5 \text{ car ils n'ont pas le même reste.}$
	
	
	\item $-15 \not\equiv -2 \pmod 4 \quad \text{car} \quad -15 = 4 \cdot (-4) + \boxed{1} \quad \text{et} \quad -2 = 4 \cdot (-1) + \boxed{2}$
	
	$-15 \text{ n'est pas congru à } -2 \text{ modulo } 4 \text{ car ils n'ont pas le même reste.}$
\end{exemples}


\begin{proposition}
~

\label{thm-cong}

\vspace{-8mm}
\begin{enumerate}
  \item La relation \og congru modulo $n$ \fg{} est une relation d'équivalence :
  
  \vspace{-1mm}
  \begin{enumerate}[label=\roman*),left=0mm]
    \item (Réflexivité) $a \equiv a \pmod n$,
    \item (Symétrie) si $a \equiv b \pmod n$ alors $b \equiv a \pmod n$,
    \item (Transitivité) si $a \equiv b \pmod n$ et $b \equiv c \pmod n$ alors $a \equiv c \pmod n$.
  \end{enumerate}
  \item $a \equiv b \pmod n \iff a - b \equiv 0 \pmod n \iff n \mid (a - b) \iff a = nk + b \quad k \in \mathbb{Z}$
  
  \item Si $a \equiv b \pmod n$ et $c \equiv d \pmod n$ alors  la congruence est compatible avec :
  
  \begin{enumerate}[label=\roman*),left=0mm]
  	\item (l’addition) :  $a+c \equiv b+d \pmod n$,
  	\item (la multiplication) : $a\cdot c \equiv b\cdot d \pmod n$,
  	\item (l'exponentiation) :  pour tout $k \geqslant 0$\; on a\; $a^k \equiv b^k \pmod n$.
  \end{enumerate}
  
\end{enumerate}
\end{proposition}



\vspace{4mm}
\noindent{\it Démonstration:}
\begin{enumerate}
  \item \textbf{1.i)} A voir que : $a \equiv a \pmod n \quad (\text{\textit{réflexive}})$
  
  Evident : Il existe des nombres $q \text{ et } r$ uniques tel que : $a = nq + r \quad \text{avec } 0 \le r < n$
  
  \vspace{0.3cm}
  
  \textbf{1.ii)} A voir que : Si $a \equiv b  \pmod n$ alors $b \equiv a  \pmod n \quad (\text{\textit{symétrique}})$
  
  Evident : Si $a \equiv b  \pmod n$ alors existe des nombres $q, q' \text{ et } r$ uniques tel que :
  
  $$a = nq + r \quad \text{et} \quad b = nq' + r \quad \text{avec } 0 \le r < n \quad \text{donc } b \equiv a  \pmod n$$
  
  \vspace{0.3cm}
  
  \textbf{1.iii)} A voir que : Si $a \equiv b  \pmod n$ et $b \equiv c  \pmod n$ alors $a \equiv c  \pmod n \quad (\text{\textit{transitive}})$
  
  $$\text{Si } a \equiv b  \pmod n \implies \begin{cases} a = nq + r & 0 \le r < n \\ b = nq' + r & 0 \le r < n \end{cases} \quad r \text{ unique}$$
  
  $$\text{et } b \equiv c  \pmod n \implies \begin{cases} b = nq' + r & 0 \le r < n \\ c = nq'' + r & 0 \le r < n \end{cases} \quad r \text{ unique}$$
  
  On constate que $a$ et $c$ ont le même reste lorsque l'on effectue une division euclidienne par $n$ d'où $a \equiv c  \pmod n$.
  \item  En Utilisant la définition,
  \begin{align*}
  	a \equiv b  \pmod n &\iff \begin{cases} a = nq + r \\ b = nq' + r \end{cases} \quad q \text{ et } q' \in \mathbb{Z} \text{ et } 0 < r \le n \\
  	&\iff a - b = (nq + r) - (nq' + r)\\
  	&\iff a - b = n \underbrace{(q - q')}_{\in \mathbb{Z}} + 0\\
  	&\iff a - b = n \underbrace{(q - q')}_{\in \mathbb{Z}} + 0\\
  	&\iff n \mid (a - b)\\
  	&\iff a - b \equiv 0  \pmod n\\
  	&\iff a - b = nk + 0 \quad k \in \mathbb{Z}\\
  	&\iff a = nk + b \quad k \in \mathbb{Z}
  \end{align*}

  \item i) Par hypothèse :
  
  $$a \equiv b  \pmod n \quad \text{et} \quad c \equiv d  \pmod n \overset{P2)}{\iff} \exists\, k, k' \in \mathbb{Z} \text{ tel que : } a = kn + b \quad \text{et} \quad c = k'n + d .$$
  
  \indent En additionnant ces deux égalités, on obtient :  
  $a + c = \underbrace{(k + k')}_{\in \mathbb{Z}} n + b + d \iff a + c \equiv b + d  \pmod n$\\
  
  \smallskip
  
  ii) Soit $a$ et $b$ tels que $a \equiv b \pmod n$, donc il existe $k \in \Z$ tel que $a=b + kn$.
  
%  \smallskip
   \indent Soit $c$ et $d$ tels que $c \equiv d \pmod n$, donc il existe $\ell \in \Z$ tel que $c \equiv d + \ell n$.
   
 \indent \indent Alors $a\fois c = (b+kn)\fois(d+\ell n) = bd + (b \ell + d k + k\ell n)\fois n$ qui est bien de la forme
$bd + m n$ avec $m\in \Z$. On conclut donc que  $a\fois c \equiv b\fois d \pmod n$.

   \smallskip
    iii) C'est une conséquence du point précédent : avec $a=c$ et $b=d$ on obtient
$a^2 \equiv b^2 \pmod n$. On continue par récurrence.
\end{enumerate}

\vspace{5mm}

La compatibilité de la relation de congruence avec l'addition, la multiplication et l’exponentiation permet de grandement simplifier les calculs de reste.

\vspace{5mm}

\begin{exemples}{1mm}
	\item  On a $50^{100} \equiv 1 \pmod 7$.
	 
	 En effet, comme  $50 \equiv 1 \pmod 7$ (car $50 = 7  \cdot 7 + 1$).	\\
	 De plus, d'après la compatibilité avec les puissances, on a : $50^{100} \equiv 1^{100} \equiv 1 \pmod 7$.
	
	
	\vspace{0.3cm}
	
	\item  On a $100^{3} \equiv 1 \pmod 7$.
	
	 En effet, comme $100 \equiv 2 \pmod 7$ (car $100 \equiv 2\cdot 50 \equiv 2\cdot 1 \pmod 7$), d'après la compatibilité avec les puissances, on a :
	$100^3 \equiv 2^3 \equiv 8 \equiv 1 \pmod 7$.
	
	\vspace{0.3cm}
	
	\item  On a $50^{100} + 100^{100} \equiv  3 \pmod 7$.
	
	 En effet, comme $100^{100} = 100^{3  \cdot 33 + 1} = \left(100^3\right)^{33}  \cdot 100$, donc d'après la compatibilité avec les puissances et la multiplication, on a : $100^{100} \equiv 1^{33} \cdot 2 \equiv 2 \pmod 7$.
	
	D'après la compatibilité avec l'addition, on a alors : $50^{100} + 100^{100} \equiv 1 + 2 \equiv 3 \pmod 7$.
	
	Le reste est 3.
\end{exemples}

\vspace{5mm}

De même, on peut utiliser ces relations pour déterminer si on nombre est un diviseur (ou un multiple) d'un autre.

\vspace{5mm}

\begin{exemple}
	On peut montrer que $\forall n \in \mathbb{N}$, le nombre $3^{n+3}-4^{4 n+2}$  est divisible par 11.
	
	\smallskip \smallskip
	
	
	En effet, on a : $3^{n+3} = 3^n \cdot 3^3 = 27 \cdot 3^n$.
	
	Or $27 \equiv 5 \pmod{11}$, donc d'après la compatibilité avec la multiplication, on a :
	
	$$\forall n \in \mathbb{N}, \quad 3^{n+3} \equiv 5 \cdot 3^n \pmod{11}$$
	
	On a : $4^{4n+2} = \left(4^4\right)^n \cdot 4^2$, or $4^2 \equiv 5 \pmod{11}$ donc $4^4 \equiv 5^2 \equiv 3 \pmod{11}$, donc :
	
	$$\forall n \in \mathbb{N}, \quad 4^{4n+2} \equiv 3^n \cdot 5 \pmod{11}$$
	
	On en déduit donc que :
	
	$$3^{n+3} - 4^{4n+2} \equiv 0 \pmod{11}$$
	
	La proposition est donc vérifiée pour tout entier naturel $n$.
\end{exemple}

\subsection{L’ensemble des entiers modulo n}

\begin{definition}
\ \vspace{-0.5cm}
\begin{enumerate}
	\item On appelle \textbf{classe de congruence} de $a$ modulo $n$ l'ensemble de tous les entiers qui sont congrus à $a$ modulo $n$ ; on la note $[a]_n$ ou $\overline{a}_n$ (ou simplement $\overline{a}$ s'il n'y a aucune ambiguïté sur $n$).
	$$[a]_n = \overline{a}_n = \{b \in \mathbb{Z} \mid a \equiv b \bmod n\} = \{a + kn \mid k \in \mathbb{Z}\}$$
	
	\item On note $\mathbb{Z}_n$ l'ensemble de toutes les classes modulo $n$ :
	$$\mathbb{Z}_n = \{[0]_n; [1]_n; [2]_n; [3]_n; \dots; [n-2]_{n-1}; [n-1]_n\}$$
\end{enumerate}

\end{definition}

\medskip

Essentiellement on regroupe dans la même classe de congruence tous les entiers qui aurons le même reste dans la division entière ou dit autrement tous les entiers qui sont séparés d'un multiple du diviseurs.

\medskip

\begin{exemple}
	L'ensemble $\mathbb{Z}_5$ est composé des 5 éléments suivants :
	\begin{align*}
		[0]_5 &= \{\dots; -5; 0; 5; 10; 15; \dots\} = \{x \in \mathbb{Z} \mid x \equiv 0 \bmod 5\} \\
		[1]_5 &= \{\dots; -4; 1; 6; 11; 16; \dots\} = \{x \in \mathbb{Z} \mid x \equiv 1 \bmod 5\} \\
		[2]_5 &= \{\dots; -3; 2; 7; 12; 17; \dots\} = \{x \in \mathbb{Z} \mid x \equiv 2 \bmod 5\} \\
		[3]_5 &= \{\dots; -2; 3; 8; 13; 18; \dots\} = \{x \in \mathbb{Z} \mid x \equiv 3 \bmod 5\} \\
		[4]_5 &= \{\dots; -1; 4; 9; 14; 19; \dots\} = \{x \in \mathbb{Z} \mid x \equiv 4 \bmod 5\}
	\end{align*}

\end{exemple}

	\item L’élément dans les crochet est un \textbf{représentant} de l’ensemble, c’est un nombre quelconque de l’ensemble (par exemple $[1]_5=[6]_5=[-4]_5\neq [3]_5$ ) bien qu’à l’usage on préfère choisir des représentant entre 0 et n-1, dans ce cas on parle de \textbf{représentant principal}.

La proposition \ref{thm-cong}, nous permet de définir une multiplication et une addition $\mathbb{Z}_.$ 

\begin{definition}
	%	\ \vspace{-0.5cm}
Soit $\cl{a}$ et $\cl{b} \in \mathbb{Z}_n$, alors on peut définir deux opérations internes sur $\mathbb{Z}_n$ :

\begin{enumerate}[label=\textbf{\arabic*)}, leftmargin=*]
	\item une \textbf{addition} : $\cl{a} + \cl{b} := \cl{a + b}$ \hfill (\text{on calcule le reste de } $a + b \bmod n$)
	\item une \textbf{multiplication} : $\cl{a} \cdot \cl{b} := \cl{a \cdot b}$ \hfill (\text{on calcule le reste de } $a \cdot b \bmod n$)
\end{enumerate}
	
\end{definition}

\medskip

\begin{remarque}
	La proposition \ref{thm-cong} nous assure que les opérations sont bien définies car elles ne dépendent pas du choix des représentants des classes utilisés. En effet, prenons deux couples qui représentent la même classe, par exemple $a \equiv b \pmod n$ et $c \equiv d \pmod n$, la proposition \ref{thm-cong} nous assure que  $a+c \equiv b+d \pmod n$ et  $a\cdot c \equiv b\cdot d \pmod n$, ce qui veut dire que $\cl{a+b}=\cl{c+d}$ et $\cl{a\cdot  b}=\cl{c\cdot d}$	
\end{remarque}

En s'appuyant des propriétés de l'addition et de la multiplication des nombres entier, ainsi que la proposition \ref{thm-cong}, on peut démontrer les propriétés suivantes:
\begin{proposition}
	~
	
	Les \textbf{\textit{entiers modulo $n$}} jouissent des \textbf{\textit{propriétés}} ci-dessous, c'est-à-dire que quelles que soient les entiers $\cl{a}, \cl{b}$ et $\cl{c} \in \mathbb{Z}_n$, les relations suivantes sont toujours vraies :
	
	\begin{description}[font=\normalfont\bfseries, leftmargin=1.2cm, style=nextline]
		\item[P1)] Si $\cl{a}, \cl{b} \in \mathbb{Z}_n$ alors $\cl{a} + \cl{b} \in \mathbb{Z}_n$. \hfill \textit{L'addition est une opération interne.}
		\item[P2)] $\cl{a} + \cl{b} = \cl{b} + \cl{a}$ \hfill \textit{Commutativité de l'addition.}
		\item[P3)] $\cl{a} + (\cl{b} + \cl{c}) = (\cl{a} + \cl{b}) + \cl{c}$ \hfill \textit{Associativité de l'addition.}
		\item[P4)] $\cl{0} + \cl{a} = \cl{a} + \cl{0} = \cl{a}$ \hfill \textit{Existence d'un élément neutre pour l'addition.}
		\item[P5)] $\forall \cl{a} \in \mathbb{Z}_n, \exists\,\text{unique } \cl{b} \in \mathbb{Z}_n \text{ tel que } \cl{a} + \cl{b} = \cl{b} + \cl{a} = \cl{0}$ \hfill \textit{Existence d'un unique opposé.}
		\item[P6)] Si $\cl{a}, \cl{b} \in \mathbb{Z}_n$ alors $\cl{a} \cdot \cl{b} \in \mathbb{Z}_n$. \hfill \textit{La multiplication est une opération interne.}
		\item[P7)] $\cl{a} \cdot \cl{b} = \cl{b} \cdot \cl{a}$ \hfill \textit{Commutativité de la multiplication.}
		\item[P8)] $\cl{a} \cdot (\cl{b} \cdot \cl{c}) = (\cl{a} \cdot \cl{b}) \cdot \cl{c}$ \hfill \textit{Associativité de la multiplication.}
		\item[P9)] $\cl{a} \cdot \cl{1} = \cl{1} \cdot \cl{a} = \cl{a}$ \hfill \textit{Existence d'un élément neutre pour la multiplication.}
		\item[P10)] $\cl{a} \cdot (\cl{b} + \cl{c}) = \cl{a} \cdot \cl{b} + \cl{a} \cdot \cl{c}$ \hfill \textit{La multiplication est distributive sur l'addition.}
	\end{description}
\end{proposition}

\medskip

\begin{remarques}
	\item Les propriétés 1) à 5) précisent que $\mathbb{Z}_n$ est une structure algébrique appelée : \\
	\textbf{groupe commutatif} vis-à-vis de l'addition et on note alors $(\mathbb{Z}_n\, ; +)$.
	
	\item Les propriétés 1) à 10) précisent que $\mathbb{Z}_n$ est un \textbf{anneau commutatif} et on note alors $(\mathbb{Z}_n\, ; +, \cdot)$.
	
	\item Si $\forall \cl{a} \neq \cl{0} \in \mathbb{Z}_n, \, \exists\,\text{\textit{unique}} \, \cl{b} \in \mathbb{Z}_n \text{ tel que } \cl{a} \cdot \cl{b} = \cl{b} \cdot \cl{a} = \cl{1} \quad (\textit{existence d'un unique inverse})$ \\
	alors $\mathbb{Z}_n$ est appelé \textbf{un corps}. \\
	Exemple : $(\mathbb{Z}_3\, ; +, \cdot)$ est un corps mais pas $(\mathbb{Z}_4\, ; +, \cdot)$.
\end{remarques}

\medskip

Dans $(\mathbb{Z}$, la soustraction ou la division ne sont pas de opérations proprement dites, il en est de même dans $(\mathbb{Z}_n$, on parle plutôt d'addition par l'inverse pour la multiplication, et de multiplication pas l'inverse (lorsqu'il existe) pour la division.

\medskip

\begin{definition}
%	\ \vspace{-0.5cm}
Soit $\cl{a}, \cl{b} \in \mathbb{Z}_n$.

\begin{enumerate}[label=\textbf{\arabic*)}, leftmargin=*]
	\item $\cl{b}$ est \textbf{l'opposé} de $\cl{a}$ si $\cl{a} + \cl{b} = \cl{b} + \cl{a} = \cl{0}$ \hfill ($a + b \equiv b + a \equiv 0 \pmod n$)
	\item $\cl{b}$ est \textbf{l'inverse} de $\cl{a}$ si $\cl{a} \cdot \cl{b} = \cl{b} \cdot \cl{a} = \cl{1}$ \hfill ($a \cdot b \equiv b \cdot a \equiv 1 \pmod n$)\\
	Notation: Si $\cl{a}$ possède un inverse on le note $\cl{a}^{-1}$.
\end{enumerate}
	
\end{definition}

\begin{remarques}
	\item Dans $\mathbb{Z}_n$, chaque élément possède un opposé. En effet, il trivial de vérifier que $\cl{-a}$ est l'opposé de $\cl{a}$.
	
	\item Dans $\mathbb{Z}_n$, il peut y avoir des éléments qui ne possède pas d'inverse.
	
\end{remarques}
